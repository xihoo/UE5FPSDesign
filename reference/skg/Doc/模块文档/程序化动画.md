  ```cpp
  USTRUCT(BlueprintType)
  struct FSKGCurve
  {
      GENERATED_BODY()
      UPROPERTY(EditDefaultsOnly, BlueprintReadWrite, Category = "SKGProcedural")
      TObjectPtr<UCurveVector> LocationCurve;
      UPROPERTY(EditDefaultsOnly, BlueprintReadWrite, Category = "SKGProcedural")
      TObjectPtr<UCurveVector> RotationCurve;
      UPROPERTY(EditDefaultsOnly, BlueprintReadWrite, Category = "SKGProcedural")
      float PlayRate {4.0f};
      UPROPERTY(EditDefaultsOnly, BlueprintReadWrite, Category = "SKGProcedural")
      float CurveEndTime {1.0f};

      bool GetVectorValues(const float Time, FVector& Location, FRotator& Rotation, bool bOffhandIKIsLeftHand)
      {
          bool bValid = false;
          if (LocationCurve)
          {
              Location = LocationCurve->GetVectorValue(Time);
              Location.X *= bOffhandIKIsLeftHand ? 1.0 : -1.0;
              bValid = true;
          }
          if (RotationCurve)
          {
              const FVector RotationVector = RotationCurve->GetVectorValue(Time);
              Rotation = FRotator(RotationVector.X, RotationVector.Y, RotationVector.Z);
              Rotation.Pitch *= bOffhandIKIsLeftHand ? 1.0 : -1.0;
              Rotation.Yaw *= bOffhandIKIsLeftHand ? 1.0 : -1.0;
              bValid = true;
          }
          return bValid;
      }
  };
  ```

  - 精确控制: 通过曲线可以精确控制动画的每一个时间点的参数

  ### 4. 核心逻辑分析

  #### 4.1 动画选择逻辑

  GetActiveChildIndex() 方法是该节点的核心逻辑，负责根据当前激活的标签选择对应的动画：

  ```cpp
  int32 FSKGAnimNode_GameplayTagBlend::GetActiveChildIndex()
  {
      // 检查标签是否发生变化
      if (ActiveGameplayTag != PreviousGameplayTag && ActiveGameplayTag.IsValid())
      {
          // 将标签容器转换为数组
          TArray<FGameplayTag> GameplayTags;
          GameplayTagContainer.GetGameplayTagArray(GameplayTags);

          // 查找当前标签在数组中的位置，并加1（因为索引0是默认动画）
          CurrentIndex = GameplayTags.Find(ActiveGameplayTag) + 1;
      }
      else
      {
          // 如果没有有效标签，使用默认动画（索引0）
          CurrentIndex = 0;
      }

      return CurrentIndex;
  }
  ```

  逻辑分析：
  1. 标签变化检测: 只有当 ActiveGameplayTag 与 PreviousGameplayTag 不同且有效时才重新计算
  2. 索引映射: 将标签在容器中的位置映射到动画索引（+1是因为索引0保留给默认动画）
  3. 默认处理: 当没有有效标签时，返回索引0（默认动画）
  4. 性能优化: 通过缓存 PreviousGameplayTag 避免不必要的计算

  #### 4.2 编辑器支持逻辑

  RefreshPoses() 方法提供了编辑器中的动态姿势管理：

  ```cpp
  #if WITH_EDITOR
  void FSKGAnimNode_GameplayTagBlend::RefreshPoses()
  {
      // 获取标签数组
      TArray<FGameplayTag> GameplayTags;
      GetTags().GetGameplayTagArray(GameplayTags);

      // 计算标签数量与动画姿势数量的差异
      const int32 TagCount = BlendPose.Num() - GameplayTags.Num() - 1;

      // 如果数量匹配，直接返回
      if (TagCount == 0)
      {
          return;
      }

      // 如果动画姿势过多，删除多余的
      if (TagCount > 0)
      {
          for (int32 i = TagCount; i > 0; i--)
          {
              RemovePose(BlendPose.Num() - 1);
          }
      }
      // 如果动画姿势不足，添加新的
      else
      {
          for (int32 i = TagCount; i < 0; i++)
          {
              AddPose();
          }
      }
  }
  #endif
  ```

  编辑器逻辑分析：
  1. 自动同步: 根据标签数量自动调整动画姿势数量
  2. 动态调整: 支持添加和删除动画姿势
  3. 用户体验: 简化了设计师的工作流程，无需手动管理动画姿势

  ### 8. 限制与注意事项

  #### 8.1 标签顺序依赖

  - 文档要求: 需要清晰的文档说明标签与动画的映射关系

  当没有匹配标签时，使用索引0的默认动画：


  设计考虑：

  - 配置要求: 需要确保索引0有一个合适的默认动画
  - 行为一致性: 所有未匹配的标签都使用同一个默认动画
