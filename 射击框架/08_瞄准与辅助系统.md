# 射击框架 · 瞄准与辅助系统

## 瞄准插值

```cpp
void UFPSShooterPawnComponent::UpdateAiming(float Dt)
{
    float Target = bIsAiming ? 1.f : 0.f;
    AimingProgress = FMath::FInterpTo(AimingProgress, Target, Dt, AimSpeed);
    if (APlayerController* PC = Cast<APlayerController>(GetOwner()->GetInstigatorController()))
    {
        if (APlayerCameraManager* CM = PC->PlayerCameraManager)
        {
            float TargetFOV = bIsAiming ? AimedFOV : DefaultFOV;
            CM->SetFOV(FMath::Lerp(DefaultFOV, TargetFOV, AimingProgress));
        }
    }
    UpdateWeaponPosition();
}
```

## 瞄准辅助（手柄）

```cpp
void UFPSShooterPawnComponent::ApplyAimAssist(FVector& OutLookDir)
{
    if (!bEnableAimAssist || !IsUsingGamepad()) return;
    AActor* NearestEnemy = FindNearestEnemy(AimAssistRange);
    if (NearestEnemy)
    {
        FVector ToEnemy = (NearestEnemy->GetActorLocation() - GetOwner()->GetActorLocation()).GetSafeNormal();
        float Dot = FVector::DotProduct(OutLookDir, ToEnemy);
        if (Dot > AimAssistAngle)
        {
            OutLookDir = FMath::VInterpTo(OutLookDir, ToEnemy, GetWorld()->GetDeltaSeconds(), AimAssistStrength);
        }
    }
}
```
